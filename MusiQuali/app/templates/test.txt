        const ENTREPRISE_ID = parseInt("{{ session['idEntreprise'] }}");

        // --- GESTION DU BOUTON ENREGISTRER (Dirty State) ---
        let isDirty = false;
        function markDirty() {
            isDirty = true;
            document.getElementById('save-planning-btn').style.display = 'block';
        }
        function clearDirty() {
            isDirty = false;
            document.getElementById('save-planning-btn').style.display = 'none';
        }

        // --- ONGLETS ---
        function switchTab() {
            const val = document.getElementById("tabSelect").value;
            document.getElementById("tab-playlists").classList.add("hidden-tab");
            document.getElementById("tab-messages").classList.add("hidden-tab");
            document.getElementById("tab-upload").classList.add("hidden-tab");
            document.getElementById(val).classList.remove("hidden-tab");
        }

        // --- GESTION DES BOÎTES DANS LE CALENDRIER ---
        let boxCounter = 0;

        function addDurationToTime(timeStr, durationSeconds) {
            const [h, m] = timeStr.split(':').map(Number);
            let totalMinutes = h * 60 + m + Math.ceil(durationSeconds / 60);
            
            const rh = Math.floor(totalMinutes / 60).toString().padStart(2, '0');
            const rm = (totalMinutes % 60).toString().padStart(2, '0');
            return `${rh}:${rm}`;
        }
        
        function createBox(item, options = {}) {
            boxCounter++;
            const { idPlanning = null, time = '07:00', dayId, type } = options;
            
            const duration = parseInt(item.duree_totale || item.dureeTotale || item.duree || 0);
            const endTime = addDurationToTime(time, duration);

            const newBox = document.createElement('div');
            newBox.id = `box-${boxCounter}`;
            newBox.className = 'scheduled-box ' + (type === 'playlists' ? 'commercial-box' : 'message-calendar-box');
            newBox.dataset.time = time;
            newBox.dataset.endTime = endTime;
            newBox.dataset.idPlanning = idPlanning || '';

            if (type === 'playlists') {
                newBox.dataset.idPlaylist = item.idPlaylist;
                newBox.innerHTML = `
                    <div class="box-header"><span><b>${time} - ${endTime}</b></span> <button class="delete-box-btn" onclick="deleteBox(this)">&times;</button></div>
                    <div class="box-title">${item.title}</div>
                    <div class="box-details-link" onclick="openDetailsModal('${item.idPlaylist}', '${item.title.replace(/'/g, "\\'")}')">+ Détails</div>
                `;
            } else {
                newBox.dataset.idMusique = item.idMusique;
                newBox.innerHTML = `
                    <div class="box-header"><span><b>${time} - ${endTime}</b></span> <button class="delete-box-btn" onclick="deleteBox(this)">&times;</button></div>
                    <div class="box-title">${item.nomMusique}</div>
                `;
            }

            document.getElementById(dayId).appendChild(newBox);
            sortColumnByTime(document.getElementById(dayId));
            return newBox;
        }

        function deleteBox(btnElement) {
            const box = btnElement.closest('.scheduled-box');
            box.remove();
            markDirty();
        }

        // Trie les boîtes par ordre chronologique
        function sortColumnByTime(column) {
            const boxes = Array.from(column.children).filter(el => el.classList.contains('scheduled-box'));
            boxes.sort((a, b) => (a.dataset.time || '99:99').localeCompare(b.dataset.time || '99:99'));
            boxes.forEach(box => column.appendChild(box));
        }

        // --- LOGIQUE DES MODALES ---
        let currentAddDayId = null;
        let currentAddType = null;
        let fetchedData = [];

        function closeModals() {
            document.querySelectorAll('.modal-overlay').forEach(m => m.classList.remove('open'));
        }

        function getEarliestTime(dayId) {
            const boxes = Array.from(document.getElementById(dayId).querySelectorAll('.scheduled-box'));
            if (boxes.length === 0) return "07:00"; 
            
            boxes.sort((a, b) => (a.dataset.time || '00:00').localeCompare(b.dataset.time || '00:00'));
            const lastBox = boxes[boxes.length - 1];
            
            return lastBox.dataset.endTime || "07:00";
        }

        async function openAddModal(dayId, type) {
            currentAddDayId = dayId;
            currentAddType = type;
            
            document.getElementById('add-modal-title').textContent = type === 'playlists' ? 'Programmer une playlist' : 'Programmer un message';
            document.getElementById('add-time-input').value = getEarliestTime(dayId);
            
            const listContainer = document.getElementById('add-modal-list');
            listContainer.innerHTML = '<i>Chargement...</i>';
            document.getElementById('add-modal').classList.add('open');

            try {
                const url = type === 'playlists' ? '/marketing/playlists' : '/marketing/musiques?prefix=MSG_';
                const res = await fetch(url);
                fetchedData = await res.json();

                if (fetchedData.length === 0) {
                    listContainer.innerHTML = '<i>Aucun élément disponible.</i>';
                    return;
                }

                listContainer.innerHTML = '<div class="radio-list"></div>';
                const radioGroup = listContainer.querySelector('.radio-list');

                fetchedData.forEach((item, index) => {
                    const label = type === 'playlists' ? item.title : item.nomMusique;
                    const html = `

                            <input type="radio" name="prog_selection" value="${index}" ${index === 0 ? 'checked' : ''}>

                    `;
                    radioGroup.insertAdjacentHTML('beforeend', html);

        function confirmAdd() {
            const selectedRadio = document.querySelector('input[name="prog_selection"]:checked');
            if (!selectedRadio) return closeModals();

            const item = fetchedData[selectedRadio.value];

        }

        async function openDetailsModal(playlistId, playlistTitle) {
            document.getElementById('details-modal-title').innerHTML = `Composition de la playlist <b>${playlistTitle}</b>`;
            const listContainer = document.getElementById('details-modal-list');
            listContainer.innerHTML = '<li><i>Chargement...</i></li>';
            document.getElementById('details-modal').classList.add('open');

            try {
                const res = await fetch(`/marketing/playlist/${playlistId}/musiques`, { headers: { 'X-Requested-With': 'XMLHttpRequest' }});
                const musiques = await res.json();
                
                if (musiques.length === 0) {
                    listContainer.innerHTML = '<li><i>Playlist vide</i></li>';
                    return;
                }
                
                listContainer.innerHTML = '';
                musiques.forEach(m => {
                    listContainer.insertAdjacentHTML('beforeend', `<li>${m.nomMusique}</li>`);
                });
            } catch (err) {
                listContainer.innerHTML = '<li><i>Erreur de chargement</i></li>';
            }
        }

        // --- SAUVEGARDE ET CHARGEMENT DU PLANNING EN BDD ---
        const DAY_OFFSETS = { 'mon': 0, 'tue': 1, 'wed': 2, 'thu': 3, 'fri': 4, 'sat': 5, 'sun': 6 };

        function getCurrentMonday() {
            const now = new Date();
            const day = now.getDay(); 
            const diffToMonday = (day === 0) ? -6 : 1 - day;
            return new Date(now.getFullYear(), now.getMonth(), now.getDate() + diffToMonday);
        }

        function buildStartTime(dayId, time) {
            if (!time) return null;
            const dayCode = dayId.split('-')[0];
            const offset = DAY_OFFSETS[dayCode];
            if (offset === undefined) return null;

            const target = getCurrentMonday();
            target.setDate(target.getDate() + offset);
            const [hours, minutes] = time.split(':').map(Number);
            target.setHours(hours, minutes, 0, 0);

            const pad = (n) => String(n).padStart(2, '0');
            return `${target.getFullYear()}-${pad(target.getMonth() + 1)}-${pad(target.getDate())}T${pad(target.getHours())}:${pad(target.getMinutes())}:00`;
        }

        function startTimeToDayAndTime(startTime, suffix) {
            const dt = new Date(startTime);
            const diffDays = Math.round((new Date(dt.getFullYear(), dt.getMonth(), dt.getDate()) - getCurrentMonday()) / 86400000);
            const dayCodes = ['mon', 'tue', 'wed', 'thu', 'fri', 'sat', 'sun'];
            const dayCode = dayCodes[((diffDays % 7) + 7) % 7];
            const pad = (n) => String(n).padStart(2, '0');
            return { dayId: `${dayCode}-${suffix}`, time: `${pad(dt.getHours())}:${pad(dt.getMinutes())}` };
        }

        async function savePlanning() {
            const saveStatus = document.getElementById('save-status');
            saveStatus.textContent = 'Sauvegarde en cours...';

            const boxes = [];
            document.querySelectorAll('.scheduled-box').forEach(box => {
                const time = box.dataset.time;
                const dayId = box.parentElement.id;
                const startTime = buildStartTime(dayId, time);
                if (!startTime) return;

                boxes.push({
                    idPlanning: box.dataset.idPlanning ? parseInt(box.dataset.idPlanning) : null,
                    idPlaylist: box.dataset.idPlaylist ? parseInt(box.dataset.idPlaylist) : null,
                    idMSG: box.dataset.idMusique ? parseInt(box.dataset.idMusique) : null,
                    StartTime: startTime,
                    _box: box
                });
            });

            const payloadBoxes = boxes.map(({ _box, ...rest }) => rest);

            try {
                const res = await fetch('/planning/save', {
                    method: 'POST',
                    headers: { 'Content-Type': 'application/json' },
                    body: JSON.stringify({ idEntreprise: 1, boxes: payloadBoxes })
                });

                if (res.ok) {
                    const saved = await res.json();
                    saved.forEach((row, i) => { if (boxes[i]._box) boxes[i]._box.dataset.idPlanning = row.idPlanning; });
                    
                    saveStatus.textContent = 'Génération des fichiers de configuration...';
                    
                    setTimeout(async () => {
                        saveStatus.textContent = 'Déploiement vers les Raspberry...';
                        
                        try {
                            const forceRes = await fetch('/admin/api/force_send_planning', { method: 'POST' });
                            if (forceRes.ok) {
                                saveStatus.textContent = 'Déploiement réussi ! ✓';
                            } else {
                                saveStatus.textContent = 'Erreur lors du déploiement.';
                            }
                        } catch (err) {
                            saveStatus.textContent = 'Erreur réseau transfert.';
                        }
                        
                        setTimeout(() => saveStatus.textContent = '', 3000);
                    }, 5000);

                    clearDirty(); 
                } else {
                    saveStatus.textContent = 'Erreur lors de la sauvegarde.';
                }
            } catch (err) { saveStatus.textContent = 'Erreur réseau.'; }
        }

        async function loadPlanning() {
            try {
                const [planningRes, playlistsRes, musiquesRes] = await Promise.all([
                    fetch(`/planning/data?idEntreprise=${ENTREPRISE_ID}`),
                    fetch('/marketing/playlists'),
                    fetch('/marketing/musiques?prefix=MSG_')
                ]);

                const rows = await planningRes.json();
                const playlists = playlistsRes.ok ? await playlistsRes.json() : [];
                const messages = musiquesRes.ok ? await musiquesRes.json() : [];

                const playlistById = new Map(playlists.map(p => [p.idPlaylist, p]));
                const messageById = new Map(messages.map(m => [m.idMusique, m]));

                // On vide les calendriers existants avant de les reconstruire
                document.querySelectorAll('.scheduled-box').forEach(b => b.remove());
                boxCounter = 0;

                for (const row of rows) {
                    if (row.idPlaylist !== null) {
                        const { dayId, time } = startTimeToDayAndTime(row.StartTime, 'tab1');
                        const p = playlistById.get(row.idPlaylist) || { idPlaylist: row.idPlaylist, title: `Playlist #${row.idPlaylist}` };
                        createBox(p, { idPlanning: row.idPlanning, time, dayId, type: 'playlists' });
                    } else if (row.idMSG !== null) {
                        const { dayId, time } = startTimeToDayAndTime(row.StartTime, 'tab2');
                        const m = messageById.get(row.idMSG) || { idMusique: row.idMSG, nomMusique: `Message #${row.idMSG}` };
                        createBox(m, { idPlanning: row.idPlanning, time, dayId, type: 'messages' });
                    }
                }
                clearDirty();
            } catch (err) { console.error(err); }
        }

        // =========================================================
        // --- SYSTÈME DE RAFRAÎCHISSEMENT TEMPS RÉEL (ASYNCHRONE) ---
        // =========================================================
        
        let currentPlanningSignature = "";

        async function fetchPlanningSignature() {
            try {
                const res = await fetch(`/planning/data?idEntreprise=${ENTREPRISE_ID}`, { headers: { 'X-Requested-With': 'XMLHttpRequest' }});
                return await res.text(); 
            } catch (e) {
                return null;
            }
        }

        setInterval(async () => {
            // SÉCURITÉ : Ne rien faire si une modification est en attente (isDirty = true)
            // Ou si une fenêtre modale d'ajout/détail est ouverte !
            const isModalOpen = document.querySelectorAll('.modal-overlay.open').length > 0;
            if (isDirty || isModalOpen) return;

            const sig = await fetchPlanningSignature();
            
            if (sig && currentPlanningSignature && sig !== currentPlanningSignature) {
                currentPlanningSignature = sig;
                console.log("Mise à jour distante détectée : Rechargement du calendrier...");
                await loadPlanning();
            } else if (sig && !currentPlanningSignature) {
                currentPlanningSignature = sig;
            }
        }, 5000);

        // Interception de la sauvegarde pour mettre à jour la signature locale sans recharger
        const originalSave = savePlanning;
        savePlanning = async function() {
            await originalSave(); 
            setTimeout(async () => {
                const sig = await fetchPlanningSignature();
                if (sig) currentPlanningSignature = sig;
            }, 500);
        };


        // ==========================================
        // 2. MOTEUR D'UPLOAD DE MESSAGES (Intact)
        // ==========================================

        const audioInputMessage = document.getElementById('audio-input-message');
        const filePreviewContainerMessage = document.getElementById('file-preview-container-message');
        const btnUploadMessage = document.getElementById('btn-upload-message');
        const uploadTextMessage = document.getElementById('upload-text-message');
        const messagesList = document.getElementById('messages-list');

        let selectedMessageFilesTransfer = new DataTransfer();

        function withMsgPrefix(file) {
            const alreadyPrefixed = file.name.startsWith('MSG_');
            const newName = alreadyPrefixed ? file.name : `MSG_${file.name}`;
            return new File([file], newName, { type: file.type });
        }

        if (audioInputMessage) {
            audioInputMessage.addEventListener('change', function() {
                for (let i = 0; i < this.files.length; i++) {
                    selectedMessageFilesTransfer.items.add(withMsgPrefix(this.files[i]));
                }
                this.files = selectedMessageFilesTransfer.files;
                updateMessageFilePreview();
            });
        }

        function updateMessageFilePreview() {
            filePreviewContainerMessage.innerHTML = ''; 

            if (selectedMessageFilesTransfer.files.length > 0) {
                filePreviewContainerMessage.classList.remove('hidden');
                btnUploadMessage.disabled = false; 
                uploadTextMessage.innerText = "Ajouter d'autres fichiers (.mp3)";

                Array.from(selectedMessageFilesTransfer.files).forEach((file, index) => {
                    const fileItem = document.createElement('div');
                    fileItem.className = 'file-preview-item';

                    const fileName = document.createElement('span');
                    fileName.className = 'item-name';
                    fileName.innerText = file.name;

                    const removeBtn = document.createElement('span');
                    removeBtn.className = 'remove-file-btn';
                    removeBtn.innerHTML = '&times;';

                    removeBtn.addEventListener('click', function(e) {
                        e.preventDefault(); 
                        removeMessageFile(index);
                    });

                    fileItem.appendChild(fileName);
                    fileItem.appendChild(removeBtn);
                    filePreviewContainerMessage.appendChild(fileItem);
                });
            } else {
                filePreviewContainerMessage.classList.add('hidden');
                btnUploadMessage.disabled = true;
                uploadTextMessage.innerText = "Glisser-déposer ou cliquer pour sélectionner un fichier";
            }
        }

        function removeMessageFile(indexToRemove) {
            const newTransfer = new DataTransfer();
            Array.from(selectedMessageFilesTransfer.files).forEach((file, index) => {
                if (index !== indexToRemove) {
                    newTransfer.items.add(file);
                }
            });
            selectedMessageFilesTransfer = newTransfer;
            audioInputMessage.files = selectedMessageFilesTransfer.files;
            updateMessageFilePreview(); 
        }

        function createMessageListBox(m) {
            const box = document.createElement('div');
            box.id = `upload-message-box-${m.idMusique}`;
            box.className = 'message-box';
            box.dataset.musicId = m.idMusique;
            box.innerHTML = `
                <span class="item-info">
                    <b>${m.nomMusique}</b>
                    <span style="white-space: nowrap;">— ${m.duree}s</span>
                </span>
                <form action="/marketing/music/${m.idMusique}/delete" method="POST" style="margin: 0;">
                    <button type="submit" class="action-btn">&times;</button>
                </form>
            `;

            box.querySelector('form').addEventListener('submit', async (ev) => {
                ev.preventDefault();
                const f = ev.target;
                const b = f.closest('.message-box');
                try {
                    const r = await fetch(f.action, {
                        method: 'POST',
                        headers: { 'X-Requested-With': 'XMLHttpRequest' }
                    });
                    if (r.ok) {
                        b.remove();
                    } else {
                        console.error('Erreur lors de la suppression', r.status);
                    }
                } catch (err) {
                    console.error('Erreur réseau', err);
                }
            });

            if(messagesList) messagesList.appendChild(box);
        }

        async function loadExistingMessages() {
            try {
                const response = await fetch('/marketing/musiques?prefix=MSG_');
                if (!response.ok) throw new Error('Erreur de chargement');
                const messages = await response.json();
                messages.forEach(m => createMessageListBox(m));
            } catch (err) {
                console.error(err);
            }
        }

        if (messagesList) {
            loadExistingMessages();
        }

        const uploadFormMessage = document.getElementById('upload-form-message');
        if (uploadFormMessage) {
            uploadFormMessage.addEventListener('submit', async (e) => {
                e.preventDefault();
                const formData = new FormData(uploadFormMessage);

                try {
                    const res = await fetch(uploadFormMessage.action, {
                        method: 'POST',
                        headers: { 'X-Requested-With': 'XMLHttpRequest' },
                        body: formData
                    });

                    if (!res.ok) {
                        console.error('Erreur lors de l\'upload', res.status);
                        return;
                    }

                    const data = await res.json();
                    (data.musiques || []).forEach(m => createMessageListBox(m));

                    selectedMessageFilesTransfer = new DataTransfer();
                    audioInputMessage.value = '';
                    updateMessageFilePreview();

                } catch (err) {
                    console.error('Erreur réseau', err);
                }
            });
        }

        // ==========================================
        // 3. LANCEMENT AU CHARGEMENT DE LA PAGE
        // ==========================================
        switchTab();
        loadPlanning();
    </script>
